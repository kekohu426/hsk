#### 1.3 数据库查询
```sql
-- 聚合查询用户学习数据
SELECT 
  u.id,
  u.username,
  u.email,
  COUNT(uw.id) as totalWords,
  SUM(CASE WHEN uw.status = 'MASTERED' THEN 1 ELSE 0 END) as masteredWords,
  SUM(CASE WHEN uw.status = 'LEARNING' THEN 1 ELSE 0 END) as learningWords,
  u.streakDays,
  SUM(ls.duration) as totalLearningTime,
  COUNT(ls.id) as totalReviews,
  AVG(CASE WHEN uw.correctCount + uw.wrongCount > 0 
      THEN uw.correctCount * 100.0 / (uw.correctCount + uw.wrongCount) 
      ELSE 0 END) as accuracy
FROM users u
LEFT JOIN user_words uw ON u.id = uw.userId
LEFT JOIN learning_sessions ls ON u.id = ls.userId
GROUP BY u.id
ORDER BY u.lastActiveAt DESC;
```

#### 3.1 修改 UserLearningTools 组件

**文件：** `frontend-user/components/word/UserLearningTools.tsx`

```typescript
export function UserLearningTools({ word, userProgress, onUpdate }: Props) {
  const [isInWordBank, setIsInWordBank] = useState(!!userProgress);
  const [status, setStatus] = useState(userProgress?.status || 'NEW');
  const [isFavorite, setIsFavorite] = useState(userProgress?.isFavorite || false);
  
  // 添加到词库
  const handleAddToWordBank = async () => {
    try {
      await api.post('/api/user/words', { 
        wordId: word.id,
        source: 'Word Detail'
      });
      setIsInWordBank(true);
      setStatus('NEW');
      toast.success('已加入词库');
      onUpdate();
    } catch (error) {
      toast.error('添加失败');
    }
  };
  
  // 更新状态
  const handleUpdateStatus = async (newStatus: 'LEARNING' | 'MASTERED') => {
    if (!isInWordBank) {
      toast.error('请先加入词库');
      return;
    }
    
    try {
      await api.patch(`/api/user/words/${userProgress.id}`, {
        status: newStatus
      });
      setStatus(newStatus);
      toast.success(`已标记为${newStatus === 'MASTERED' ? '已掌握' : '学习中'}`);
      onUpdate();
    } catch (error) {
      toast.error('更新失败');
    }
  };
  
  // 切换收藏
  const handleToggleFavorite = async () => {
    if (!isInWordBank) {
      toast.error('请先加入词库');
      return;
    }
    
    try {
      await api.patch(`/api/user/words/${userProgress.id}`, {
        isFavorite: !isFavorite
      });
      setIsFavorite(!isFavorite);
      toast.success(isFavorite ? '已取消收藏' : '已收藏');
      onUpdate();
    } catch (error) {
      toast.error('操作失败');
    }
  };
  
  // 移除词汇
  const handleRemove = async () => {
    if (!confirm('确定要从词库中移除这个词汇吗？')) return;
    
    try {
      await api.delete(`/api/user/words/${userProgress.id}`);
      setIsInWordBank(false);
      toast.success('已移除');
      onUpdate();
    } catch (error) {
      toast.error('移除失败');
    }
  };
  
  return (
    <Card className="mb-6">
      <CardContent className="p-6">
        <div className="flex flex-wrap items-center gap-3">
          {/* 如果未加入词库，显示添加按钮 */}
          {!isInWordBank ? (
            <Button 
              onClick={handleAddToWordBank}
              className="bg-blue-600 hover:bg-blue-700"
            >
              <BookmarkPlus className="w-4 h-4 mr-2" />
              加入词库
            </Button>
          ) : (
            <>
              {/* 状态按钮 */}
              <Button
                onClick={() => handleUpdateStatus('MASTERED')}
                variant={status === 'MASTERED' ? 'default' : 'outline'}
                className={status === 'MASTERED' ? 'bg-green-600 hover:bg-green-700' : ''}
              >
                <CheckCircle className="w-4 h-4 mr-2" />
                {status === 'MASTERED' ? '✓ 已掌握' : '标记为已掌握'}
              </Button>
              
              <Button
                onClick={() => handleUpdateStatus('LEARNING')}
                variant={status === 'LEARNING' ? 'default' : 'outline'}
                className={status === 'LEARNING' ? 'bg-orange-600 hover:bg-orange-700' : ''}
              >
                <BookOpen className="w-4 h-4 mr-2" />
                {status === 'LEARNING' ? '✓ 学习中' : '加入学习'}
              </Button>
              
              {/* 收藏按钮 */}
              <Button
                onClick={handleToggleFavorite}
                variant={isFavorite ? 'default' : 'outline'}
                className={isFavorite ? 'bg-yellow-500 hover:bg-yellow-600' : ''}
              >
                <Star className={`w-4 h-4 mr-2 ${isFavorite ? 'fill-white' : ''}`} />
                {isFavorite ? '已收藏' : '收藏'}
              </Button>
              
              {/* 移除按钮 */}
              <Button
                onClick={handleRemove}
                variant="outline"
                className="text-red-600 border-red-300 hover:bg-red-50"
              >
                <Trash2 className="w-4 h-4 mr-2" />
                移除
              </Button>
            </>
          )}
        </div>
        
        {/* 学习进度信息 */}
        {isInWordBank && userProgress && (
          <div className="mt-4 flex items-center gap-6 text-sm text-gray-600">
            <div>
              <span className="font-medium">当前状态:</span>{' '}
              <Badge variant={
                status === 'MASTERED' ? 'success' : 
                status === 'LEARNING' ? 'warning' : 'default'
              }>
                {status === 'MASTERED' ? '已掌握' : 
                 status === 'LEARNING' ? '学习中' : '新词汇'}
              </Badge>
            </div>
            <div>复习次数: {userProgress.repetitions}</div>
            <div>
              正确率: {userProgress.correctCount + userProgress.wrongCount > 0
                ? Math.round(userProgress.correctCount * 100 / 
                    (userProgress.correctCount + userProgress.wrongCount))
                : 0}%
            </div>
          </div>
        )}
      </CardContent>
    </Card>
  );
}
```
