# 任务队列系统设计

## 📋 模块概述

**目标**: 实现基于BullMQ的任务队列系统，支持批量AI生成、进度监控、失败重试。

**核心功能**:
- 批量添加生成任务到队列
- 并发处理任务（控制并发数）
- 实时进度监控
- 失败自动重试（最多3次）
- 任务优先级管理

---

## 🏗️ 技术架构

### 技术栈
- **队列系统**: BullMQ
- **Redis**: Upstash Redis（Serverless Redis）
- **Worker**: 独立进程或Vercel Cron Jobs

### 架构图
```
┌────────────────┐
│  管理后台API   │
│  触发批量生成  │
└───────┬────────┘
        │ 创建任务
        ↓
┌────────────────┐
│  GenerationTask│
│  数据库记录    │
└───────┬────────┘
        │
        ↓
┌────────────────┐
│   BullMQ队列   │
│  (Redis存储)   │
└───────┬────────┘
        │
        ↓
┌────────────────┐
│  Worker进程    │
│  (并发处理)    │
└───────┬────────┘
        │
        ├─→ AI服务 → 生成内容
        ├─→ 更新状态
        └─→ 失败重试
```

## 🚀 任务队列服务

### 队列管理服务
**文件**: `api/services/taskQueueService.js`

```javascript
const { Queue } = require('bullmq');
const connection = require('../config/redis');
const { PrismaClient } = require('@prisma/client');
const prisma = new PrismaClient();

class TaskQueueService {
  constructor() {
    // 创建队列实例
    this.generationQueue = new Queue('generation-queue', {
      connection,
      defaultJobOptions: {
        attempts: 3,
        backoff: {
          type: 'exponential',
          delay: 5000  // 5秒后第一次重试
        },
        removeOnComplete: {
          age: 24 * 3600,  // 24小时后删除已完成任务
          count: 1000      // 保留最近1000个
        },
        removeOnFail: {
          age: 7 * 24 * 3600  // 7天后删除失败任务
        }
      }
    });
  }

  /**
   * 添加单个生成任务到队列
   */
  async addGenerationTask(taskId, type, targetId, priority = 0) {
    try {
      const job = await this.generationQueue.add(
        `${type}-${targetId}`,
        {
          taskId,
          type,
          targetId
        },
        {
          priority,
          jobId: `task-${taskId}`
        }
      );

      console.log(`[TaskQueue] Task ${taskId} added to queue: ${job.id}`);
      return job;

    } catch (error) {
      console.error(`[TaskQueue] Failed to add task ${taskId}:`, error);
      throw error;
    }
  }

  /**
   * 批量添加任务到队列
   */
  async addBatchToQueue(batchId, tasks) {
    try {
      const jobs = tasks.map(task => ({
        name: `${task.type}-${task.targetId}`,
        data: {
          taskId: task.id,
          type: task.type,
          targetId: task.targetId,
          batchId
        },
        opts: {
          priority: task.priority || 0,
          jobId: `task-${task.id}`
        }
      }));

      await this.generationQueue.addBulk(jobs);

      console.log(`[TaskQueue] Batch ${batchId}: ${jobs.length} tasks added`);
      return { success: true, count: jobs.length };

    } catch (error) {
      console.error(`[TaskQueue] Failed to add batch ${batchId}:`, error);
      throw error;
    }
  }

  /**
   * 获取队列状态
   */
  async getQueueStats() {
    const [
      waiting,
      active,
      completed,
      failed,
      delayed
    ] = await Promise.all([
      this.generationQueue.getWaitingCount(),
      this.generationQueue.getActiveCount(),
      this.generationQueue.getCompletedCount(),
      this.generationQueue.getFailedCount(),
      this.generationQueue.getDelayedCount()
    ]);

    return {
      waiting,
      active,
      completed,
      failed,
      delayed,
      total: waiting + active + completed + failed + delayed
    };
  }

  /**
   * 获取批次进度
   */
  async getBatchProgress(batchId) {
    try {
      // 从数据库获取批次任务统计
      const tasks = await prisma.generationTask.findMany({
        where: { batchId },
        select: {
          id: true,
          status: true,
          targetId: true,
          type: true,
          createdAt: true,
          startedAt: true,
          completedAt: true,
          errorMessage: true
        }
      });

      const stats = {
        total: tasks.length,
        pending: tasks.filter(t => t.status === 'PENDING').length,
        processing: tasks.filter(t => t.status === 'PROCESSING').length,
        completed: tasks.filter(t => t.status === 'COMPLETED').length,
        failed: tasks.filter(t => t.status === 'FAILED').length
      };

      stats.percentage = stats.total > 0
        ? Math.round((stats.completed / stats.total) * 100)
        : 0;

      // 估算剩余时间
      const completedTasks = tasks.filter(t => t.status === 'COMPLETED' && t.completedAt && t.startedAt);
      if (completedTasks.length > 0) {
        const avgTime = completedTasks.reduce((sum, t) => {
          return sum + (t.completedAt.getTime() - t.startedAt.getTime());
        }, 0) / completedTasks.length;

        const remainingTasks = stats.pending + stats.processing;
        const estimatedMs = remainingTasks * avgTime;
        stats.estimatedTimeRemaining = this.formatDuration(estimatedMs);
      } else {
        stats.estimatedTimeRemaining = 'Calculating...';
      }

      // 获取最近完成的任务
      const recentTasks = tasks
        .filter(t => t.status === 'COMPLETED')
        .sort((a, b) => b.completedAt - a.completedAt)
        .slice(0, 5)
        .map(t => ({
          taskId: t.id,
          type: t.type,
          targetId: t.targetId,
          completedAt: t.completedAt
        }));

      return {
        batchId,
        stats,
        recentTasks,
        startedAt: tasks[0]?.createdAt
      };

    } catch (error) {
      console.error(`[TaskQueue] Failed to get batch progress:`, error);
      throw error;
    }
  }

  /**
   * 暂停队列
   */
  async pauseQueue() {
    await this.generationQueue.pause();
    console.log('[TaskQueue] Queue paused');
  }

  /**
   * 恢复队列
   */
  async resumeQueue() {
    await this.generationQueue.resume();
    console.log('[TaskQueue] Queue resumed');
  }

  /**
   * 取消批次中的所有待处理任务
   */
  async cancelBatch(batchId) {
    try {
      // 1. 检查批次是否存在
      const batch = await prisma.generationBatch.findUnique({
        where: { id: batchId },
        include: { tasks: true }
      });

      if (!batch) {
        throw new Error('批次不存在');
      }

      // 2. 检查批次是否已完成（无法取消已完成的批次）
      if (batch.status === 'COMPLETED') {
        throw new Error('批次已完成，无法取消');
      }

      // 3. 统计任务状态
      const statusCount = {
        pending: 0,
        processing: 0,
        completed: 0,
        failed: 0
      };

      batch.tasks.forEach(task => {
        if (task.status === 'PENDING') statusCount.pending++;
        else if (task.status === 'PROCESSING') statusCount.processing++;
        else if (task.status === 'COMPLETED') statusCount.completed++;
        else if (task.status === 'FAILED') statusCount.failed++;
      });

      // 4. 取消所有PENDING和PROCESSING状态的任务
      const tasksToCancel = batch.tasks.filter(
        task => task.status === 'PENDING' || task.status === 'PROCESSING'
      );

      // 更新数据库中的任务状态
      await prisma.generationTask.updateMany({
        where: {
          batchId,
          status: { in: ['PENDING', 'PROCESSING'] }
        },
        data: {
          status: 'CANCELLED'
        }
      });

      // 从队列中移除对应的Job
      let removedJobs = 0;
      for (const task of tasksToCancel) {
        try {
          const job = await this.generationQueue.getJob(`task-${task.id}`);
          if (job && (await job.getState()) !== 'completed') {
            await job.remove();
            removedJobs++;
          }
        } catch (error) {
          console.error(`Failed to remove job for task ${task.id}:`, error);
        }
      }

      // 5. 更新批次状态
      await prisma.generationBatch.update({
        where: { id: batchId },
        data: { status: 'CANCELLED' }
      });

      const result = {
        success: true,
        batchId,
        cancelledTasks: tasksToCancel.length,
        removedJobs,
        completedTasks: statusCount.completed,
        failedTasks: statusCount.failed,
        breakdown: {
          pending: statusCount.pending,
          processing: statusCount.processing
        }
      };

      console.log(`[TaskQueue] Batch ${batchId} cancelled:`, result);
      return result;

    } catch (error) {
      console.error(`[TaskQueue] Failed to cancel batch ${batchId}:`, error);
      throw error;
    }
  }

  /**
   * 格式化持续时间
   */
  formatDuration(ms) {
    const seconds = Math.floor(ms / 1000);
    const minutes = Math.floor(seconds / 60);
    const hours = Math.floor(minutes / 60);

    if (hours > 0) {
      return `${hours}h ${minutes % 60}m`;
    } else if (minutes > 0) {
      return `${minutes}m ${seconds % 60}s`;
    } else {
      return `${seconds}s`;
    }
  }

  /**
   * 清理旧任务
   */
  async cleanOldJobs() {
    await this.generationQueue.clean(24 * 3600 * 1000, 1000, 'completed');
    await this.generationQueue.clean(7 * 24 * 3600 * 1000, 1000, 'failed');
    console.log('[TaskQueue] Old jobs cleaned');
  }
}

module.exports = new TaskQueueService();
```

## 👷 Worker进程实现

**文件**: `api/workers/generationWorker.js`

```javascript
const { Worker } = require('bullmq');
const connection = require('../config/redis');
const aiService = require('../services/aiService');
const articleAIService = require('../services/articleAIService');
const { PrismaClient } = require('@prisma/client');
const prisma = new PrismaClient();

// 创建Worker实例
const generationWorker = new Worker(
  'generation-queue',
  async (job) => {
    const { taskId, type, targetId, batchId } = job.data;

    console.log(`\n[Worker] 🚀 Starting task ${taskId}`);
    console.log(`[Worker] Type: ${type}, Target: ${targetId}, Batch: ${batchId || 'N/A'}`);

    try {
      // 1. 更新任务状态为处理中
      await prisma.generationTask.update({
        where: { id: taskId },
        data: {
          status: 'PROCESSING',
          startedAt: new Date()
        }
      });

      // 2. 根据类型调用相应的生成服务
      let result;
      const startTime = Date.now();

      if (type === 'WORD_ENTRY') {
        console.log(`[Worker] Generating word entry...`);
        result = await aiService.generateWordEntry(targetId);
      } else if (type === 'ARTICLE') {
        console.log(`[Worker] Generating article...`);
        result = await articleAIService.generateArticle(targetId);
      } else {
        throw new Error(`Unknown task type: ${type}`);
      }

      const duration = ((Date.now() - startTime) / 1000).toFixed(2);
      console.log(`[Worker] ✅ Task ${taskId} completed in ${duration}s`);

      // 3. 更新任务状态为已完成
      await prisma.generationTask.update({
        where: { id: taskId },
        data: {
          status: 'COMPLETED',
          completedAt: new Date()
        }
      });

      return result;

    } catch (error) {
      console.error(`[Worker] ❌ Task ${taskId} failed:`, error.message);

      // 获取当前任务信息
      const task = await prisma.generationTask.findUnique({
        where: { id: taskId }
      });

      const attempts = (job.attemptsMade || 0) + 1;
      const maxAttempts = task.maxAttempts || 3;

      if (attempts < maxAttempts) {
        // 还有重试机会
        console.log(`[Worker] 🔄 Will retry task ${taskId} (attempt ${attempts}/${maxAttempts})`);

        await prisma.generationTask.update({
          where: { id: taskId },
          data: {
            status: 'PENDING',
            attempts,
            errorMessage: error.message
          }
        });

      } else {
        // 达到最大重试次数
        console.log(`[Worker] 💀 Task ${taskId} permanently failed after ${maxAttempts} attempts`);

        await prisma.generationTask.update({
          where: { id: taskId },
          data: {
            status: 'FAILED',
            attempts,
            errorMessage: error.message
          }
        });
      }

      throw error;
    }
  },
  {
    connection,
    concurrency: 3,  // 同时处理3个任务
    limiter: {
      max: 10,       // 每10秒最多处理10个任务
      duration: 10000
    },
    settings: {
      stalledInterval: 300000,  // 5分钟检查一次停滞任务
      maxStalledCount: 1        // 停滞1次后标记为失败
    }
  }
);

// 监听Worker事件
generationWorker.on('completed', (job, result) => {
  console.log(`[Worker] ✓ Job ${job.id} completed successfully`);
});

generationWorker.on('failed', (job, err) => {
  console.error(`[Worker] ✗ Job ${job.id} failed:`, err.message);
});

generationWorker.on('error', (err) => {
  console.error('[Worker] Worker error:', err);
});

generationWorker.on('stalled', (jobId) => {
  console.warn(`[Worker] ⚠ Job ${jobId} has stalled`);
});

// 优雅关闭
process.on('SIGTERM', async () => {
  console.log('[Worker] SIGTERM received, closing worker...');
  await generationWorker.close();
  await connection.quit();
  process.exit(0);
});

process.on('SIGINT', async () => {
  console.log('[Worker] SIGINT received, closing worker...');
  await generationWorker.close();
  await connection.quit();
  process.exit(0);
});

console.log('[Worker] Generation worker started and ready');

module.exports = generationWorker;
```

    }
  ]

**文件**: `api/cron/process-queue.js`

```javascript
const taskQueueService = require('../services/taskQueueService');
const aiService = require('../services/aiService');
const { PrismaClient } = require('@prisma/client');
const prisma = new PrismaClient();

module.exports = async (req, res) => {
  // 验证Cron密钥
  if (req.headers.authorization !== `Bearer ${process.env.CRON_SECRET}`) {
    return res.status(401).json({ error: 'Unauthorized' });
  }

  try {
    // 获取待处理的任务（限制每次处理5个）
    const pendingTasks = await prisma.generationTask.findMany({
      where: {
        status: 'PENDING'
      },
      take: 5,
      orderBy: [
        { priority: 'desc' },
        { createdAt: 'asc' }
      ]
    });

    if (pendingTasks.length === 0) {
      return res.json({ message: 'No pending tasks' });
    }

    // 处理任务
    const results = [];
    for (const task of pendingTasks) {
      try {
        await prisma.generationTask.update({
          where: { id: task.id },
          data: { status: 'PROCESSING', startedAt: new Date() }
        });

        if (task.type === 'WORD_ENTRY') {
          await aiService.generateWordEntry(task.targetId);
        }

        await prisma.generationTask.update({
          where: { id: task.id },
          data: { status: 'COMPLETED', completedAt: new Date() }
        });

        results.push({ taskId: task.id, status: 'success' });

      } catch (error) {
        const attempts = task.attempts + 1;

        await prisma.generationTask.update({
          where: { id: task.id },
          data: {
            status: attempts < task.maxAttempts ? 'PENDING' : 'FAILED',
            attempts,
            errorMessage: error.message
          }
        });

        results.push({ taskId: task.id, status: 'failed', error: error.message });
      }
    }

    res.json({
      processed: pendingTasks.length,
      results
    });

  } catch (error) {
    console.error('[Cron] Error:', error);
    res.status(500).json({ error: error.message });
  }
};
```

## 🔧 故障排查

### 问题1：Worker无法启动
**检查**:
- Redis连接是否正常
- 环境变量REDIS_URL是否正确
- 端口是否被占用

### 问题2：任务一直pending
**检查**:
- Worker进程是否在运行
- 查看Worker日志是否有错误
- 检查任务状态: `await queue.getJob('task-123')`

### 问题3：任务重复执行
**检查**:
- jobId是否唯一
- removeOnComplete配置是否正确
