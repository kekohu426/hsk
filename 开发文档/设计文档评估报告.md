class APIClient {
  constructor() {
    this.cache = new Map();
    this.offlineQueue = [];
  }

  // 支持离线的GET请求
  async getWithCache(endpoint, options = {}) {
    const cacheKey = `${endpoint}?${JSON.stringify(options.params)}`;

    // 尝试从缓存读取
    if (this.cache.has(cacheKey)) {
      const cached = this.cache.get(cacheKey);
      if (Date.now() - cached.timestamp < 3600000) { // 1小时有效
        return cached.data;
      }
    }

    // 检查网络状态
    if (!navigator.onLine) {
      if (this.cache.has(cacheKey)) {
        return this.cache.get(cacheKey).data;
      }
      throw new Error('离线状态，且无缓存数据');
    }

    // 从网络获取
    const data = await this.get(endpoint, options);

    // 写入缓存
    this.cache.set(cacheKey, {
      data,
      timestamp: Date.now(),
    });

    return data;
  }

  // 离线队列：POST/PUT请求在离线时加入队列
  async postWithOfflineSupport(endpoint, data) {
    if (!navigator.onLine) {
      this.offlineQueue.push({
        method: 'POST',
        endpoint,
        data,
        timestamp: Date.now(),
      });
      return { offline: true, message: '已加入离线队列' };
    }

    return await this.post(endpoint, data);
  }

  // 恢复在线时同步队列
  async syncOfflineQueue() {
    if (this.offlineQueue.length === 0) return;

    console.log(`同步 ${this.offlineQueue.length} 个离线请求...`);

    for (const request of this.offlineQueue) {
      try {
        if (request.method === 'POST') {
          await this.post(request.endpoint, request.data);
        } else if (request.method === 'PUT') {
          await this.put(request.endpoint, request.data);
        }
      } catch (error) {
        console.error('离线请求同步失败:', error);
      }
    }

    this.offlineQueue = [];
  }
}
