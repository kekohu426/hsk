**新建文件**: `api-client.js`

```javascript
// API基础配置
const API_BASE_URL = window.location.hostname === 'localhost'
  ? 'http://localhost:3000/api'
  : 'https://your-api.vercel.app/api';

// API客户端类
class APIClient {
  constructor() {
    this.baseURL = API_BASE_URL;
    this.token = localStorage.getItem('auth_token');

    // 离线缓存相关
    this.cache = new Map();
    this.offlineQueue = this.loadOfflineQueue();
    this.setupOnlineListener();
  }

  // 加载离线队列
  loadOfflineQueue() {
    try {
      const stored = localStorage.getItem('offline_queue');
      return stored ? JSON.parse(stored) : [];
    } catch (error) {
      console.error('Failed to load offline queue:', error);
      return [];
    }
  }

  // 保存离线队列
  saveOfflineQueue() {
    try {
      localStorage.setItem('offline_queue', JSON.stringify(this.offlineQueue));
    } catch (error) {
      console.error('Failed to save offline queue:', error);
    }
  }

  // 监听网络状态变化
  setupOnlineListener() {
    window.addEventListener('online', () => {
      console.log('[APIClient] Network online, syncing offline queue...');
      this.syncOfflineQueue();
    });

    window.addEventListener('offline', () => {
      console.log('[APIClient] Network offline');
    });
  }

  // 检查网络状态
  isOnline() {
    return navigator.onLine;
  }

  // 设置token
  setToken(token) {
    this.token = token;
    if (token) {
      localStorage.setItem('auth_token', token);
    } else {
      localStorage.removeItem('auth_token');
    }
  }

  // 获取token
  getToken() {
    return this.token || localStorage.getItem('auth_token');
  }

  // 通用请求方法
  async request(endpoint, options = {}) {
    const url = `${this.baseURL}${endpoint}`;
    
    const config = {
      ...options,
      headers: {
        'Content-Type': 'application/json',
        ...options.headers
      }
    };

    // 添加认证token
    const token = this.getToken();
    if (token) {
      config.headers['Authorization'] = `Bearer ${token}`;
    }

    try {
      const response = await fetch(url, config);
      
      if (!response.ok) {
        if (response.status === 401) {
          // Token过期，跳转登录
          this.setToken(null);
          window.location.href = '/login.html';
          throw new Error('Authentication required');
        }
        
        const error = await response.json();
        throw new Error(error.error || 'API request failed');
      }

      return await response.json();
    } catch (error) {
      console.error('API Error:', error);
      throw error;
    }
  }

  // GET请求
  get(endpoint) {
    return this.request(endpoint, { method: 'GET' });
  }

  // POST请求
  post(endpoint, data) {
    return this.request(endpoint, {
      method: 'POST',
      body: JSON.stringify(data)
    });
  }

  // PUT请求
  put(endpoint, data) {
    return this.request(endpoint, {
      method: 'PUT',
      body: JSON.stringify(data)
    });
  }

  // DELETE请求
  delete(endpoint) {
    return this.request(endpoint, { method: 'DELETE' });
  }

  // 支持离线的GET请求（带缓存）
  async getWithCache(endpoint, options = {}) {
    const cacheKey = `${endpoint}?${JSON.stringify(options.params || {})}`;
    const cacheDuration = options.cacheDuration || 3600000; // 默认1小时

    // 尝试从缓存读取
    if (this.cache.has(cacheKey)) {
      const cached = this.cache.get(cacheKey);
      if (Date.now() - cached.timestamp < cacheDuration) {
        console.log(`[Cache] Hit: ${cacheKey}`);
        return cached.data;
      }
    }

    // 检查网络状态
    if (!this.isOnline()) {
      // 离线状态，返回缓存数据（即使过期）
      if (this.cache.has(cacheKey)) {
        console.warn(`[Cache] Offline, returning expired cache: ${cacheKey}`);
        return this.cache.get(cacheKey).data;
      }
      throw new Error('离线状态，且无缓存数据');
    }

    // 从网络获取
    const data = await this.get(endpoint);

    // 写入缓存
    this.cache.set(cacheKey, {
      data,
      timestamp: Date.now()
    });

    console.log(`[Cache] Stored: ${cacheKey}`);
    return data;
  }

  // 支持离线的POST请求（离线时加入队列）
  async postWithOfflineSupport(endpoint, data) {
    if (!this.isOnline()) {
      // 离线状态，加入队列
      const queueItem = {
        id: `offline_${Date.now()}_${Math.random()}`,
        method: 'POST',
        endpoint,
        data,
        timestamp: Date.now()
      };

      this.offlineQueue.push(queueItem);
      this.saveOfflineQueue();

      console.log(`[Offline] Queued POST request: ${endpoint}`);

      return {
        offline: true,
        message: '已加入离线队列，将在恢复网络后同步',
        queueId: queueItem.id
      };
    }

    // 在线状态，直接发送
    return await this.post(endpoint, data);
  }

  // 支持离线的PUT请求（离线时加入队列）
  async putWithOfflineSupport(endpoint, data) {
    if (!this.isOnline()) {
      // 离线状态，加入队列
      const queueItem = {
        id: `offline_${Date.now()}_${Math.random()}`,
        method: 'PUT',
        endpoint,
        data,
        timestamp: Date.now()
      };

      this.offlineQueue.push(queueItem);
      this.saveOfflineQueue();

      console.log(`[Offline] Queued PUT request: ${endpoint}`);

      return {
        offline: true,
        message: '已加入离线队列，将在恢复网络后同步',
        queueId: queueItem.id
      };
    }

    // 在线状态，直接发送
    return await this.put(endpoint, data);
  }

  // 同步离线队列
  async syncOfflineQueue() {
    if (this.offlineQueue.length === 0) {
      console.log('[Sync] No offline requests to sync');
      return { success: true, synced: 0 };
    }

    console.log(`[Sync] Syncing ${this.offlineQueue.length} offline requests...`);

    const results = {
      total: this.offlineQueue.length,
      success: 0,
      failed: 0,
      errors: []
    };

    // 创建队列副本，用于同步
    const queueToSync = [...this.offlineQueue];
    this.offlineQueue = [];
    this.saveOfflineQueue();

    for (const request of queueToSync) {
      try {
        if (request.method === 'POST') {
          await this.post(request.endpoint, request.data);
        } else if (request.method === 'PUT') {
          await this.put(request.endpoint, request.data);
        }

        results.success++;
        console.log(`[Sync] Success: ${request.method} ${request.endpoint}`);
      } catch (error) {
        results.failed++;
        results.errors.push({
          request,
          error: error.message
        });

        // 同步失败，重新加入队列
        this.offlineQueue.push(request);
        console.error(`[Sync] Failed: ${request.method} ${request.endpoint}`, error);
      }
    }

    // 保存未成功的请求
    this.saveOfflineQueue();

    console.log(`[Sync] Complete: ${results.success} success, ${results.failed} failed`);
    return results;
  }

  // 清除缓存
  clearCache() {
    this.cache.clear();
    console.log('[Cache] Cleared');
  }

  // 清除离线队列
  clearOfflineQueue() {
    this.offlineQueue = [];
    this.saveOfflineQueue();
    console.log('[Offline] Queue cleared');
  }

  // === 认证API ===
  async register(email, username, password) {
    const response = await this.post('/auth/register', { email, username, password });
    if (response.success && response.data.token) {
      this.setToken(response.data.token);
    }
    return response;
  }

  async login(email, password) {
    const response = await this.post('/auth/login', { email, password });
    if (response.success && response.data.token) {
      this.setToken(response.data.token);
    }
    return response;
  }

  logout() {
    this.setToken(null);
    window.location.href = '/login.html';
  }

  // === 用户API ===
  getDashboard() {
    return this.get('/user/dashboard');
  }

  getProfile() {
    return this.get('/user/profile');
  }

  updateProfile(data) {
    return this.put('/user/profile', data);
  }

  // === 词汇API ===
  getHSKWords(level, page = 1, limit = 20) {
    return this.get(`/words/hsk/${level}?page=${page}&limit=${limit}`);
  }

  getWordBySlug(slug) {
    return this.get(`/words/${slug}`);
  }

  getUserWords() {
    return this.get('/user/words');
  }

  addWord(wordEntryId, source) {
    return this.post('/user/words', { wordEntryId, source });
  }

  deleteWord(userWordId) {
    return this.delete(`/user/words/${userWordId}`);
  }

  // === 文章API ===
  getArticles(level = null, page = 1) {
    const params = new URLSearchParams({ page, limit: 12 });
    if (level) params.append('level', level);
    return this.get(`/articles?${params}`);
  }

  getArticleBySlug(slug) {
    return this.get(`/articles/${slug}`);
  }

  // === 学习API ===
  startLearningSession() {
    return this.post('/learning/session/start');
  }

  submitLearningResults(sessionId, results) {
    return this.post(`/learning/session/${sessionId}/submit`, { results });
  }
}

// 创建全局实例
const api = new APIClient();
```

## 🔐 登录状态管理

**新建文件**: `auth-check.js`

```javascript
// 全局认证检查
(function() {
  // 不需要登录的页面
  const publicPages = [
    '/',
    '/index.html',
    '/login.html',
    '/register.html',
    '/word/',  // 词汇静态页面
    '/article/',  // 文章静态页面
  ];
  
  // 检查当前页面是否需要登录
  function requiresAuth() {
    const path = window.location.pathname;
    return !publicPages.some(page => path === page || path.startsWith(page));
  }
  
  // 检查登录状态
  function checkAuth() {
    if (requiresAuth() && !api.getToken()) {
      // 保存当前页面，登录后返回
      sessionStorage.setItem('redirect_after_login', window.location.href);
      window.location.href = '/login.html';
    }
  }
  
  // 页面加载时检查
  document.addEventListener('DOMContentLoaded', checkAuth);
  
  // 更新header中的用户信息
  async function updateHeader() {
    if (api.getToken()) {
      try {
        const response = await api.getProfile();
        if (response.success) {
          const user = response.data;
          
          // 更新用户头像和名字
          const userAvatar = document.querySelector('.user-avatar');
          const userName = document.querySelector('.user-menu span');
          
          if (userAvatar) userAvatar.src = user.avatar || 'https://i.pravatar.cc/100';
          if (userName) userName.textContent = user.username;
        }
      } catch (error) {
        console.error('Failed to load user info:', error);
      }
    }
  }
  
  // 绑定logout按钮
  document.addEventListener('DOMContentLoaded', () => {
    const logoutBtn = document.querySelector('.logout-btn');
    if (logoutBtn) {
      logoutBtn.addEventListener('click', () => {
        api.logout();
      });
    }
    
    updateHeader();
  });
})();
```
