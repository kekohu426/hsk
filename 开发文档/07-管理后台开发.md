# 07 - 管理后台开发

## 📋 模块概述

本文档提供管理后台的完整实现方案，采用 **React + Vite + shadcn/ui** 技术栈，打造现代化的 SaaS 风格管理界面。

**核心功能**:
- 词汇管理（导入、编辑、批量生成）
- 文章管理（创建、编辑、AI生成）
- AI生成监控面板（实时进度）
- 用户管理（查看、统计）
- 任务队列监控
- AI配置管理

**技术栈**:
- React 18
- Vite 5
- shadcn/ui (Radix UI + Tailwind CSS)
- React Router v6
- Zustand (状态管理)
- Axios
- React Query (数据获取)

**预计开发时间**: 2-3天

import axios from 'axios';

const API_BASE_URL = import.meta.env.VITE_API_URL || 'http://localhost:3000';

class APIClient {
  constructor() {
    this.client = axios.create({
      baseURL: `${API_BASE_URL}/api`,
      timeout: 30000,
    });

    // 请求拦截器：添加JWT token
    this.client.interceptors.request.use(
      (config) => {
        const token = localStorage.getItem('admin_token');
        if (token) {
          config.headers.Authorization = `Bearer ${token}`;
        }
        return config;
      },
      (error) => Promise.reject(error)
    );

    // 响应拦截器：处理401
    this.client.interceptors.response.use(
      (response) => response,
      (error) => {
        if (error.response?.status === 401) {
          localStorage.removeItem('admin_token');
          window.location.href = '/login';
        }
        return Promise.reject(error);
      }
    );
  }

  // ==================== 认证 ====================
  async login(email, password) {
    const { data } = await this.client.post('/auth/login', { email, password });
    if (data.user.role !== 'ADMIN') {
      throw new Error('需要管理员权限');
    }
    localStorage.setItem('admin_token', data.token);
    return data;
  }

  async logout() {
    localStorage.removeItem('admin_token');
  }

  // ==================== 仪表盘 ====================
  async getDashboardStats() {
    const { data } = await this.client.get('/admin/dashboard/stats');
    return data;
  }

  // ==================== 词汇管理 ====================
  async getWordEntries(params = {}) {
    const { data } = await this.client.get('/admin/word-entries', { params });
    return data;
  }

  async importWords(csvFile) {
    const formData = new FormData();
    formData.append('file', csvFile);
    const { data } = await this.client.post('/admin/word-entries/import', formData, {
      headers: { 'Content-Type': 'multipart/form-data' },
    });
    return data;
  }

  async batchGenerateWords(wordIds) {
    const { data } = await this.client.post('/admin/generation/batch', {
      type: 'WORD',
      targetIds: wordIds,
    });
    return data;
  }

  async updateWordEntry(wordId, updates) {
    const { data } = await this.client.put(`/admin/word-entries/${wordId}`, updates);
    return data;
  }

  async deleteWordEntry(wordId) {
    const { data } = await this.client.delete(`/admin/word-entries/${wordId}`);
    return data;
  }

  async publishWord(wordId) {
    const { data } = await this.client.post(`/admin/word-entries/${wordId}/publish`);
    return data;
  }

  // ==================== 文章管理 ====================
  async getArticles(params = {}) {
    const { data } = await this.client.get('/admin/articles', { params });
    return data;
  }

  async getArticle(articleId) {
    const { data } = await this.client.get(`/admin/articles/${articleId}`);
    return data;
  }

  async createArticle(articleData) {
    const { data } = await this.client.post('/admin/articles', articleData);
    return data;
  }

  async updateArticle(articleId, updates) {
    const { data } = await this.client.put(`/admin/articles/${articleId}`, updates);
    return data;
  }

  async deleteArticle(articleId) {
    const { data } = await this.client.delete(`/admin/articles/${articleId}`);
    return data;
  }

  async generateArticleContent(articleId) {
    const { data } = await this.client.post(`/admin/articles/${articleId}/generate`);
    return data;
  }

  // ==================== 用户管理 ====================
  async getUsers(params = {}) {
    const { data } = await this.client.get('/admin/users', { params });
    return data;
  }

  async getUserLearningStats(userId) {
    const { data } = await this.client.get(`/admin/users/${userId}/stats`);

  // ==================== 任务队列 ====================

### 3.2 全局状态管理 (`src/lib/store.js`)

```javascript
import { create } from 'zustand';

// 认证状态
export const useAuthStore = create((set) => ({
  user: null,
  token: localStorage.getItem('admin_token'),
  setUser: (user) => set({ user }),
  setToken: (token) => {
    localStorage.setItem('admin_token', token);
    set({ token });
  },
  logout: () => {
    localStorage.removeItem('admin_token');
    set({ user: null, token: null });
  },
}));

// 生成任务状态
export const useGenerationStore = create((set, get) => ({
  activeBatches: [],
  currentBatch: null,

  addBatch: (batch) => set((state) => ({
    activeBatches: [...state.activeBatches, batch],
  })),

  updateBatchProgress: (batchId, progress) => set((state) => ({
    activeBatches: state.activeBatches.map((batch) =>
      batch.id === batchId ? { ...batch, ...progress } : batch
    ),
  })),

  removeBatch: (batchId) => set((state) => ({
    activeBatches: state.activeBatches.filter((batch) => batch.id !== batchId),
  })),
}));
```

### 4.1 登录页 (`src/pages/LoginPage.jsx`)

```javascript
import { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card';
import { Input } from '@/components/ui/input';
import { Label } from '@/components/ui/label';
import { Button } from '@/components/ui/button';
import { Alert, AlertDescription } from '@/components/ui/alert';
import { apiClient } from '@/lib/api';
import { useAuthStore } from '@/lib/store';

export default function LoginPage() {
  const navigate = useNavigate();
  const setUser = useAuthStore((state) => state.setUser);
  const setToken = useAuthStore((state) => state.setToken);

  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [error, setError] = useState('');
  const [loading, setLoading] = useState(false);

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError('');
    setLoading(true);

    try {
      const data = await apiClient.login(email, password);
      setUser(data.user);
      setToken(data.token);
      navigate('/dashboard');
    } catch (err) {
      setError(err.response?.data?.error || err.message || '登录失败，请检查账号密码');
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="min-h-screen flex items-center justify-center bg-gray-50 px-4">
      <Card className="w-full max-w-md">
        <CardHeader className="space-y-1">
          <CardTitle className="text-2xl font-bold">ChineseMaster 管理后台</CardTitle>
          <CardDescription>使用管理员账号登录</CardDescription>
        </CardHeader>
        <CardContent>
          <form onSubmit={handleSubmit} className="space-y-4">
            {error && (
              <Alert variant="destructive">
                <AlertDescription>{error}</AlertDescription>
              </Alert>
            )}

            <div className="space-y-2">
              <Label htmlFor="email">邮箱</Label>
              <Input
                id="email"
                type="email"
                placeholder="admin@example.com"
                value={email}
                onChange={(e) => setEmail(e.target.value)}
                required
              />
            </div>

            <div className="space-y-2">
              <Label htmlFor="password">密码</Label>
              <Input
                id="password"
                type="password"
                value={password}
                onChange={(e) => setPassword(e.target.value)}
                required
              />
            </div>

            <Button type="submit" className="w-full" disabled={loading}>
              {loading ? '登录中...' : '登录'}
            </Button>
          </form>
        </CardContent>
      </Card>
    </div>
  );
}

        title: '导入成功',

                id="file"
                type="file"
                accept=".csv"
                onChange={handleFileChange}
                disabled={uploading}
              />
              {file && (
                <div className="flex items-center text-sm text-gray-600">
                  <FileText className="w-4 h-4 mr-1" />
                  {file.name}
                </div>
              )}
            </div>
          </div>

          <Alert>
            <AlertDescription>
              <p className="font-medium mb-2">CSV文件格式要求：</p>
              <pre className="text-xs bg-gray-50 p-2 rounded">
{`word,pinyin,english,hskLevel
你好,nǐ hǎo,hello,1
学习,xué xí,study,2`}

                {result.skipped > 0 && (

            </Alert>

import { useEffect } from 'react';

  });

        <Card>
          <CardContent className="pt-6">
            <div className="flex items-center justify-between">
              <div>
                <div className="text-2xl font-bold text-yellow-600">{pendingPages}</div>
                <div className="text-sm text-gray-600">待生成</div>
              </div>
              <Clock className="w-8 h-8 text-yellow-400" />
            </div>
          </CardContent>
        </Card>

        <Card>
          <CardContent className="pt-6">
            <div className="flex items-center justify-between">
              <div>
                <div className="text-2xl font-bold text-red-600">{failedPages}</div>
                <div className="text-sm text-gray-600">生成失败</div>
              </div>
              <AlertCircle className="w-8 h-8 text-red-400" />
            </div>
          </CardContent>
        </Card>
      </div>

      {/* 生成进度 */}
      {generationProgress > 0 && (
        <Card>
          <CardHeader>
            <CardTitle>正在生成静态页面...</CardTitle>
          </CardHeader>
          <CardContent className="space-y-4">
            <Progress value={generationProgress} className="w-full" />
            <div className="flex justify-between text-sm text-gray-600">
              <span>已完成 {generatedPages}/{totalPages}</span>
              <span>{generationProgress.toFixed(1)}%</span>
            </div>
            <p className="text-sm text-gray-500">
              预计剩余时间: {Math.ceil((totalPages - generatedPages) * 3 / 60)} 分钟
              (平均 3秒/页面)
            </p>
          </CardContent>
        </Card>
      )}

      {/* SEO配置概览 */}
      <Card>
        <CardHeader>
          <CardTitle>SEO配置</CardTitle>
        </CardHeader>
        <CardContent className="space-y-3">
          <div className="grid grid-cols-2 gap-4 text-sm">
            <div className="flex items-center">
              <CheckCircle className="w-4 h-4 text-green-500 mr-2" />
              <span>Meta标签优化 (Title, Description, Keywords)</span>
            </div>
            <div className="flex items-center">
              <CheckCircle className="w-4 h-4 text-green-500 mr-2" />
              <span>Schema.org结构化数据</span>
            </div>
            <div className="flex items-center">
              <CheckCircle className="w-4 h-4 text-green-500 mr-2" />
              <span>Open Graph社交媒体标签</span>
            </div>
            <div className="flex items-center">
              <CheckCircle className="w-4 h-4 text-green-500 mr-2" />
              <span>语义化HTML5标签</span>
            </div>
            <div className="flex items-center">
              <CheckCircle className="w-4 h-4 text-green-500 mr-2" />
              <span>内部链接优化</span>
            </div>
            <div className="flex items-center">
              <CheckCircle className="w-4 h-4 text-green-500 mr-2" />
              <span>移动端响应式设计</span>
            </div>
          </div>
        </CardContent>
      </Card>

      {/* 页面列表 */}
      <Card>
        <CardHeader>
          <div className="flex items-center justify-between">
            <CardTitle>页面列表</CardTitle>
            <div className="relative w-64">
              <Search className="absolute left-3 top-1/2 transform -translate-y-1/2 w-4 h-4 text-gray-400" />
              <Input
                placeholder="搜索词汇或URL..."
                value={searchTerm}
                onChange={(e) => setSearchTerm(e.target.value)}
                className="pl-10"
              />
            </div>
          </div>
        </CardHeader>
        <CardContent>
          <div className="space-y-2">
            {filteredPages.map((page) => (
              <div
                key={page.id}
                className="flex items-center justify-between p-4 border rounded hover:bg-gray-50"
              >
                <div className="flex-1">
                  <div className="flex items-center space-x-3">
                    <span className="font-chinese font-medium text-lg">{page.word}</span>
                    <Badge variant="outline">HSK {page.hskLevel}</Badge>
                    {page.status === 'GENERATED' && (
                      <Badge className="bg-green-100 text-green-800">
                        <CheckCircle className="w-3 h-3 mr-1" />
                        已生成
                      </Badge>
                    )}
                    {page.status === 'PENDING' && (
                      <Badge className="bg-yellow-100 text-yellow-800">
                        <Clock className="w-3 h-3 mr-1" />
                        待生成
                      </Badge>
                    )}
                    {page.status === 'FAILED' && (
                      <Badge className="bg-red-100 text-red-800">
                        <AlertCircle className="w-3 h-3 mr-1" />
                        失败
                      </Badge>
                    )}
                  </div>
                  <div className="text-sm text-gray-600 mt-1">
                    /word/{page.slug}
                  </div>
                  {page.generatedAt && (
                    <div className="text-xs text-gray-500 mt-1">
                      生成时间: {new Date(page.generatedAt).toLocaleString()}
                    </div>
                  )}
                </div>
                <div className="flex space-x-2">
                  {page.status === 'GENERATED' && (
                    <Button
                      variant="outline"
                      size="sm"
                      onClick={() => window.open(`/word/${page.slug}`, '_blank')}
                    >
                      <Globe className="w-4 h-4 mr-1" />
                      预览
                    </Button>
                  )}
                  <Button
                    variant="outline"
                    size="sm"
                    onClick={() => regenerateMutation.mutate(page.wordId)}
                    disabled={regenerateMutation.isLoading}
                  >
                    <RefreshCw className="w-4 h-4 mr-1" />
                    重新生成
                  </Button>
                </div>
              </div>
            ))}
          </div>

          {filteredPages.length === 0 && (
            <div className="text-center py-12 text-gray-500">
              <Globe className="w-12 h-12 mx-auto mb-4 opacity-20" />
              <p>没有找到匹配的页面</p>
            </div>
          )}
        </CardContent>
      </Card>
    </div>
  );
}
```
