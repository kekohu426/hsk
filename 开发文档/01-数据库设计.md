# ChineseMaster 完整开发计划

## 📌 项目概述

**目标**：基于现有HTML原型，从零开发完整的HSK学习平台（前端+后端+管理后台+AI生成）

**核心原则**：
- ✅ 用户端HTML/CSS/JS基本不动，只改API对接
- ✅ 静态SEO落地页（每个词汇一个HTML文件）
- ✅ AI生成所有内容（词汇、文章）
- ✅ 任务队列系统（批量生成、进度监控）
- ✅ SaaS风格管理后台（科技感、简洁）

---

## 🏗️ 技术栈

| 模块 | 技术选型 | 说明 |
|------|----------|------|
| 用户端前端 | HTML/CSS/JS (原样保留) + Fetch API | 轻量级改造，只改数据源 |
| 静态落地页 | Node.js 脚本生成 HTML 文件 | 10,000个词汇页面 |
| 后端API | Node.js + Express + Prisma | Vercel Serverless Functions |
| 数据库 | PostgreSQL (Vercel Postgres) | 支持JSON字段 |
| 任务队列 | BullMQ + Redis (Upstash) | 批量AI生成 |
| AI服务 | GLM-4 API (智谱) | 词汇和文章生成 |
| 管理后台 | React + Vite + shadcn/ui | 现代SaaS风格 |
| 部署 | Vercel (前端+后端+静态页面) | 全部托管在Vercel |

### 表1: User（用户表）

```prisma
model User {
  id              Int       @id @default(autoincrement())
  email           String    @unique
  username        String    @unique
  password        String    // bcrypt加密

  // 个人信息
  avatar          String?
  preferredLang   String    @default("en") // en, ja, ko

  // 学习统计
  level           Int       @default(1)
  xp              Int       @default(0)
  streak          Int       @default(0)
  lastStudyDate   DateTime?

  // 考试目标
  examGoal        Json?     // {level: 4, examDate: '2025-05-15', readiness: 64}

  // 会员状态
  isPremium       Boolean   @default(false)
  premiumUntil    DateTime?

  // 时间戳
  createdAt       DateTime  @default(now())
  updatedAt       DateTime  @updatedAt

  // 关系
  userWords       UserWord[]
  learningSessions LearningSession[]
  articleViews    ArticleView[]
  dailyMissions   DailyMission[]
  achievements    UserAchievement[]

  @@index([email])
  @@index([username])
}
```

**字段说明**：
- `email`：登录邮箱
- `username`：用户名（显示在dashboard.html顶部）
- `level`：用户等级（Level 12）
- `xp`：经验值（2450 XP）
- `streak`：连续学习天数（30-day streak）
- `examGoal`：JSON存储考试目标信息
- `isPremium`：是否是付费用户

### 表2: WordEntry（词汇条目表）- 核心！

```prisma
model WordEntry {
  id              Int       @id @default(autoincrement())

  // 基础信息
  word            String    @unique    // 中文词汇，如"你好"
  slug            String    @unique    // URL slug，如"ni-hao"
  pinyin          String                // 不带声调："ni hao"
  pinyinWithTones String                // 带声调："nǐ hǎo"
  english         String                // 英文释义："Hello / Hi"
  hskLevel        Int                   // 1-6

  // AI生成的完整内容（JSON格式）
  contentJson     Json?                 // 存储完整的Quick Learn + Deep Dive + Practice数据

  // 生成状态管理
  status          WordEntryStatus @default(PENDING_IMPORT)
  aiPrompt        String?   @db.Text    // 用于生成的提示词
  aiResponse      String?   @db.Text    // AI原始响应
  aiModel         String?               // 使用的模型："glm-4"
  generateError   String?   @db.Text    // 生成错误信息

  // SEO字段
  seoTitle        String?
  seoDescription  String?
  seoKeywords     String?
  canonicalUrl    String?   @unique

  // 静态HTML文件路径
  htmlFilePath    String?               // "/word/ni-hao.html"
  htmlGenerated   Boolean   @default(false)

  // 统计字段
  viewCount       Int       @default(0)
  wordCount       Int       @default(0)  // contentJson的总字数

  // 时间戳
  importedAt      DateTime?
  generatedAt     DateTime?
  publishedAt     DateTime?
  createdAt       DateTime  @default(now())
  updatedAt       DateTime  @updatedAt

  // 关系
  userWords       UserWord[]

  @@index([slug])
  @@index([hskLevel])
  @@index([status])
}

enum WordEntryStatus {
  PENDING_IMPORT   // 待导入
  GENERATING       // AI生成中
  GENERATED        // 已生成
  PUBLISHED        // 已发布（HTML已生成）

}
```

### 表3: Article（文章表）

```prisma
model Article {
  id              Int       @id @default(autoincrement())

  // 基础信息
  title           String               // 中文标题："北京传统美食文化"
  titleEn         String?              // 英文标题："Traditional Beijing Cuisine Culture"
  slug            String    @unique    // URL: "beijing-food-culture"

  // 内容（JSON格式）
  content         Json                 // 文章正文，每个词都有标注
  summary         String?   @db.Text   // 摘要

  // 分级
  level           ArticleLevel         // BEGINNER, INTERMEDIATE, ADVANCED
  hskLevel        String               // "3-4", "4-5"

  // 元数据
  readTime        Int                  // 阅读时长（分钟）：8
  wordCount       Int                  // 字数：586
  characterCount  Int                  // 字符数：586
  newWordsCount   Int                  // 生词数：23

  // 生词列表
  newWords        Json                 // [{word, pinyin, english, hskLevel}]

  // 语法点
  grammarPoints   Json                 // [{title, pattern, meaning, examples}]

  // 讨论问题
  discussionQuestions Json             // [{chinese, english}]

  // 媒体资源
  coverImage      String?
  audioUrl        String?

  // AI生成相关
  aiPrompt        String?   @db.Text
  aiModel         String?
  generateError   String?   @db.Text

  // SEO
  seoTitle        String?
  seoDescription  String?
  seoKeywords     String?

  // 状态
  status          ArticleStatus @default(DRAFT)

  // 静态HTML
  htmlFilePath    String?
  htmlGenerated   Boolean   @default(false)

  // 统计
  viewCount       Int       @default(0)

  // 时间戳
  publishedAt     DateTime?
  createdAt       DateTime  @default(now())
  updatedAt       DateTime  @updatedAt

  // 关系
  views           ArticleView[]

  @@index([slug])
  @@index([level])
  @@index([status])
}

enum ArticleLevel {
  BEGINNER       // 初级
  INTERMEDIATE   // 中级
  ADVANCED       // 高级
}

enum ArticleStatus {
  DRAFT          // 草稿
  GENERATING     // 生成中
  PUBLISHED      // 已发布
  ARCHIVED       // 已归档
}
```

### 表4: UserWord（用户学习记录）

```prisma
model UserWord {
  id              Int       @id @default(autoincrement())

  // 关联
  userId          Int
  user            User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  wordEntryId     Int
  wordEntry       WordEntry @relation(fields: [wordEntryId], references: [id], onDelete: Cascade)

  // 学习状态
  status          WordStatus @default(LEARNING)

  // SuperMemo 2算法字段
  repetitions     Int       @default(0)      // 复习次数
  easeFactor      Float     @default(2.5)    // 难易度因子
  interval        Int       @default(0)      // 间隔天数
  nextReview      DateTime?                  // 下次复习时间

  // 学习统计
  correctCount    Int       @default(0)
  wrongCount      Int       @default(0)
  totalTime       Int       @default(0)      // 学习总时长（秒）

  // 个人笔记
  isFavorite      Boolean   @default(false)
  notes           String?   @db.Text
  source          String?                    // 来源："hsk-library", "article", "manual"

  // 时间戳
  addedAt         DateTime  @default(now())
  lastReviewedAt  DateTime?
  updatedAt       DateTime  @updatedAt

  @@unique([userId, wordEntryId])
  @@index([userId, status])
  @@index([userId, nextReview])
}

enum WordStatus {
  LEARNING       // 学习中
  MASTERED       // 已掌握
  REVIEWING      // 复习中
}
```

```prisma
  // 时间戳
```

```prisma
model Achievement {
  id              Int       @id @default(autoincrement())

  // 成就信息
```

### 表9: GenerationTask（AI生成任务队列）- 重点！

```prisma
model GenerationTask {
  id              Int       @id @default(autoincrement())

  // 任务类型
  type            TaskType               // WORD_ENTRY, ARTICLE

  // 关联ID
  targetId        Int                    // wordEntryId 或 articleId

  // 任务状态
  status          TaskStatus @default(PENDING)
  priority        Int       @default(0)  // 优先级（越大越高）

  // 执行信息
  attempts        Int       @default(0)  // 尝试次数
  maxAttempts     Int       @default(3)
  errorMessage    String?   @db.Text

  // 批次管理
  batchId         String?                // 批次ID（批量生成时使用）

  // 时间戳
  createdAt       DateTime  @default(now())
  startedAt       DateTime?
  completedAt     DateTime?
  updatedAt       DateTime  @updatedAt

  @@index([status, priority])
  @@index([batchId])
}

enum TaskType {
  WORD_ENTRY     // 词汇生成
  ARTICLE        // 文章生成
}

enum TaskStatus {
  PENDING        // 待处理
  PROCESSING     // 处理中
  COMPLETED      // 已完成
  FAILED         // 失败
  CANCELLED      // 已取消
}
```
