```javascript
class WordListManager {
  constructor() {
    this.currentPage = 1;
    this.hasMore = true;
    this.loading = false;
  }

  async loadMore() {
    if (this.loading || !this.hasMore) return;

    this.loading = true;
    showLoadingSpinner();

    try {
      const response = await apiClient.get('/words/hsk/1', {
        params: { page: this.currentPage, limit: 20 }
      });

      this.renderWords(response.data);
      this.currentPage++;
      this.hasMore = response.pagination.hasNext;
    } finally {
      this.loading = false;
      hideLoadingSpinner();
    }
  }

  setupInfiniteScroll() {
    window.addEventListener('scroll', () => {
      if (window.innerHeight + window.scrollY >= document.body.offsetHeight - 500) {
        this.loadMore();
      }
    });
  }
}
```

**功能特点**:
- 滚动到底部前500px自动加载

async shouldRegenerate(wordEntry) {
  const outputPath = path.join(this.outputDir, 'word', `${wordEntry.slug}.html`);

  try {
    const stats = await fs.stat(outputPath);
    // 如果HTML文件比数据库记录旧，需要重新生成

  } catch (error) {

    return true;

    }

### 13. 添加06文档：API请求去重 ✅

**实现方案**:
```javascript
class APIClient {
  constructor() {
    this.pendingRequests = new Map();
  }

  async request(endpoint, options = {}) {
    const requestKey = `${options.method || 'GET'}:${endpoint}`;

    // 如果相同请求正在进行，返回已存在的Promise
    if (this.pendingRequests.has(requestKey)) {
      return this.pendingRequests.get(requestKey);
    }

    // 创建新请求
    const requestPromise = this._doRequest(endpoint, options)
      .finally(() => {
        // 请求完成后移除
        this.pendingRequests.delete(requestKey);
      });

    this.pendingRequests.set(requestKey, requestPromise);
    return requestPromise;
  }
}
```

**优点**:
- 防止快速连续点击发送多个相同请求
- 节省服务器资源
- 提升应用性能

### 14. 添加08文档：数据库备份脚本 ✅

**实现方案**:
```javascript
// scripts/backup-db.js
const { exec } = require('child_process');
const fs = require('fs').promises;
const path = require('path');

async function backupDatabase() {
  const timestamp = new Date().toISOString().replace(/[:.]/g, '-');
  const backupFile = path.join(__dirname, '../backups', `db-${timestamp}.sql`);

  const command = `pg_dump ${process.env.DATABASE_URL} > ${backupFile}`;

  return new Promise((resolve, reject) => {
    exec(command, (error, stdout, stderr) => {
      if (error) {
        console.error('备份失败:', error);
        reject(error);
      } else {
        console.log(`备份成功: ${backupFile}`);
        resolve(backupFile);
      }
    });
  });
}

// Vercel Cron定时执行
module.exports = async (req, res) => {
  try {
    const backupFile = await backupDatabase();
    // 可选：上传到云存储（如AWS S3）
    res.json({ success: true, backup: backupFile });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
};
```

**Vercel Cron配置**:
```json
{
  "crons": [{
    "path": "/api/cron/backup",
    "schedule": "0 2 * * *"
  }]
}
```

**功能**:
- 每天凌晨2点自动备份
- 支持上传到云存储
- 备份文件带时间戳
